/**
 * Frame Sizing Test
 *
 * Goal:
 * Verify that `.frame({ width, height })` and `.background()` modifiers
 * interact correctly — ensuring each background paints behind the view’s
 * current layout bounds at the moment it’s applied.
 *
 * Layout order explanation:
 *
 * 1️⃣ Circle()
 *     - Draws a basic shape with intrinsic (content-based) size.
 *
 * 2️⃣ .frame({ width: 200, height: 150 })
 *     - Constrains the rectangle to exactly 200×150.
 *     - The rectangle’s layout bounds are now fixed at 200×150.
 *
 * 3️⃣ .background(Color('red').opacity(0.5))
 *     - Applies **after** the inner frame.
 *     - The red background matches the rectangle’s *current* bounds (200×150).
 *     - Red fills the 200×150 area — not any later frame you add.
 *
 * 4️⃣ .frame({ width: 400, height: 800 })
 *     - Expands the layout space to 400×800.
 *     - The red background remains 200×150 because it was applied earlier.
 *     - The 200×150 red area is now centered within the 400×800 frame.
 *
 * 5️⃣ .background(Color('blue').opacity(0.1))
 *     - Applies **after** the outer frame.
 *     - The blue background fills the *entire* 400×800 frame.
 *     - Final visual result:
 *         🔵 Blue background (400×800)
 *           🟥 Red rectangle (200×150) centered inside
 *             🟩 Green shape (200×150 content area)
 */
const FrameTestBaseline = () => {
    return (
        VStack([
            // --- Inner child ---
            Circle()
                .fill(Color('green').opacity(0.5))
                // Child fixed at 200×150
                .frame({ width: 200, height: 150 })
        ])
            // --- Middle frame (acts as parent) ---
            // Red background matches the 200×150 frame above
            .background(Color('red').opacity(0.5))

            // --- Outer frame ---
            // Expands total layout to 400×800,
            // leaving the red background centered at 200×150
            .frame({ width: 400, height: 800 })

            // --- Outer background ---
            // Blue background matches the outer frame’s full 400×800 bounds
            .background(Color('blue').opacity(0.1))
    )
};
